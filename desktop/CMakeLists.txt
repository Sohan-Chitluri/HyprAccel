cmake_minimum_required(VERSION 3.20)
project(HyprAccelStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_executable(hypraccel-studio src/main.cpp)
target_link_libraries(hypraccel-studio PRIVATE Qt6::Widgets)
enable_testing()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/theme/CMakeLists.txt)
  add_subdirectory(src/theme)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/board/CMakeLists.txt)
  add_subdirectory(src/board)
endif()
target_sources(hypraccel-studio PRIVATE src/hardware_view.cpp)
# Pin/clock configuration components. Each subdirectory defines one library
# (studio_config, studio_pinout, studio_pinmenu, studio_clock, studio_project)
# plus its own tests; they are linked in once present.
set(HYPR_COMPONENTS "config;pinout;pinmenu;clock;project;pincheck" CACHE STRING "Configuration components to build")
foreach(component ${HYPR_COMPONENTS})
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/${component}/CMakeLists.txt)
    add_subdirectory(src/${component})
  endif()
endforeach()
foreach(library studio_config studio_pinout studio_pinmenu studio_clock studio_project studio_pincheck)
  if(TARGET ${library})
    target_link_libraries(hypraccel-studio PRIVATE ${library})
  endif()
endforeach()
target_link_libraries(hypraccel-studio PRIVATE studio_theme studio_board)
target_compile_definitions(hypraccel-studio PRIVATE BOARD_CATALOG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../boards/boards.yaml")
# Live pin-conflict check (desktop/src/pincheck): the same script the editor
# server and its tests use, invoked via QProcess. Kept alongside
# BOARD_CATALOG_PATH since both point into the same boards/ tree.
target_compile_definitions(hypraccel-studio PRIVATE PIN_CHECK_SCRIPT_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../boards/codegen/check_pin_conflicts.js")
add_executable(hardware_view_test tests/hardware_view_test.cpp src/hardware_view.cpp)
set_target_properties(hardware_view_test PROPERTIES AUTOMOC ON)
target_include_directories(hardware_view_test PRIVATE src)
target_link_libraries(hardware_view_test PRIVATE Qt6::Widgets Qt6::Test studio_board studio_theme)
foreach(library studio_config studio_pinout studio_pinmenu studio_pincheck)
  if(TARGET ${library})
    target_link_libraries(hardware_view_test PRIVATE ${library})
  endif()
endforeach()
target_compile_definitions(hardware_view_test PRIVATE
    BOARD_CATALOG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../boards/boards.yaml"
    PIN_CHECK_SCRIPT_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../boards/codegen/check_pin_conflicts.js")
add_test(NAME hardware_view COMMAND hardware_view_test)
set_tests_properties(hardware_view PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
add_test(NAME startup COMMAND hypraccel-studio --smoke-test)
add_test(NAME launcher_theme COMMAND hypraccel-studio --theme-test)
set_tests_properties(startup launcher_theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
